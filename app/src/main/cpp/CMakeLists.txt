cmake_minimum_required(VERSION 3.22.1)
project(tomsmic_native C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(TM_WHISPER "build whisper.cpp" ON)
option(TM_LAME "build LAME mp3" ON)

if (TM_WHISPER)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(whisper.cpp)
    add_library(tomsmic_whisper SHARED whisper_jni.cpp)
    target_link_libraries(tomsmic_whisper whisper log android)
    target_compile_options(tomsmic_whisper PRIVATE -O3)
endif()

if (TM_LAME)
    file(GLOB LAME_SRC lame/libmp3lame/*.c)
    list(FILTER LAME_SRC EXCLUDE REGEX ".*mpglib_interface\\.c$")
    add_library(tomsmic_mp3 SHARED mp3_jni.c ${LAME_SRC})
    target_include_directories(tomsmic_mp3 PRIVATE lame_config lame/include lame/libmp3lame)
    target_compile_definitions(tomsmic_mp3 PRIVATE HAVE_CONFIG_H)
    target_compile_options(tomsmic_mp3 PRIVATE -O2 -w)
    target_link_libraries(tomsmic_mp3 log m)
endif()
